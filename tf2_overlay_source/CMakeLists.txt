cmake_minimum_required(VERSION 3.16)
project(goal3_tf2_deadlock_overlay)
set(CMAKE_CXX_STANDARD 14)
find_package(ament_cmake REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(rcutils REQUIRED)
find_package(console_bridge REQUIRED)
find_package(GTest REQUIRED)
add_library(goal3_tf2 SHARED cache.cpp buffer_core.cpp static_cache.cpp time.cpp)
set_target_properties(goal3_tf2 PROPERTIES OUTPUT_NAME tf2 LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
target_compile_definitions(goal3_tf2 PRIVATE TF2_BUILDING_DLL)
target_include_directories(goal3_tf2 PRIVATE ${tf2_INCLUDE_DIRS})
ament_target_dependencies(goal3_tf2 geometry_msgs rcutils)
target_link_libraries(goal3_tf2 console_bridge::console_bridge)
add_executable(test_buffer_upstream test_buffer.cpp)
ament_target_dependencies(test_buffer_upstream tf2_ros rclcpp geometry_msgs)
target_link_libraries(test_buffer_upstream GTest::gtest GTest::gtest_main)
